```python
        """Ingest with automatic batching and memory management"""
        
        # Add timestamp if missing
        if 'timestamp' not in event:
            event['timestamp'] = time.time()
        
        # Pack to minimize size
        packed = msgpack.packb(event, use_bin_type=True)
        
        # Check size
        if len(packed) > 1024:  # 1KB limit per event
            # Compress large events
            packed = zlib.compress(packed, level=1)
            event['compressed'] = True
        
        # Add to stream with automatic trimming
        event_id = await self.redis.xadd(
            'events:stream',
            {'data': packed},
            maxlen=self.max_stream_length,
            approximate=True
        )
        
        # Update statistics (HyperLogLog for memory efficiency)
        await self.redis.pfadd('stats:unique:sources', event.get('source_ip', ''))
        await self.redis.pfadd('stats:unique:pids', str(event.get('pid', '')))
        
        # Increment counters with expiry
        pipe = self.redis.pipeline()
        pipe.hincrby('stats:counts', event.get('event_type', 'unknown'), 1)
        pipe.expire('stats:counts', 3600)
        await pipe.execute()
        
        return event_id
    
    async def consume_batch(self) -> List[Dict]:
        """Consume events in conservative batches"""
        
        try:
            # Read with timeout
            messages = await self.redis.xread(
                {'events:stream': '$'},
                count=self.batch_size,
                block=1000  # 1 second timeout
            )
            
            if not messages:
                return []
            
            events = []
            for stream, stream_messages in messages:
                for msg_id, data in stream_messages:
                    # Unpack event
                    packed_data = data[b'data']
                    
                    # Check if compressed
                    try:
                        event = msgpack.unpackb(packed_data, raw=False)
                    except:
                        # Try decompressing
                        unpacked = zlib.decompress(packed_data)
                        event = msgpack.unpackb(unpacked, raw=False)
                    
                    events.append(event)
                    
                    # Acknowledge and remove to free memory
                    await self.redis.xdel('events:stream', msg_id)

class TokenBucket:
    """Rate limiting implementation"""
    
    def __init__(self, rate: int, capacity: int):
        self.rate = rate
        self.capacity = capacity
        self.tokens = capacity
        self.last_update = time.time()
    
    def consume(self, tokens: int = 1) -> bool:
        """Try to consume tokens"""
        now = time.time()
        elapsed = now - self.last_update
        
        # Add new tokens
        self.tokens = min(self.capacity, self.tokens + elapsed * self.rate)
        self.last_update = now
        
        # Try to consume
        if self.tokens >= tokens:
            self.tokens -= tokens
            return True
        
        return False
```

```python
        """Load agent-specific knowledge"""
        return {
            'threat_patterns': [],
            'response_playbooks': [],
            'ioc_database': set()
        }
    
    async def analyze(self, event: Dict) -> Dict:
        """Analyze event with cognitive reasoning"""
        
        start_time = time.time()
        
        # Add to context
        self.context_window.append(event)
        
        # Quick heuristic check
        if not self.requires_deep_analysis(event):
            return self.quick_decision(event)
        
        # Deep cognitive analysis
        try:
            async with asyncio.timeout(self.config.decision_timeout_seconds):
                decision = await self.cognitive_analysis(event)
        except asyncio.TimeoutError:
            decision = self.timeout_decision(event)
        
        # Track metrics
        self.update_metrics(time.time() - start_time, decision)
        
        return decision
    
    def requires_deep_analysis(self, event: Dict) -> bool:
        """Determine if LLM analysis needed"""
        
        # Conservative triggers
        triggers = [
            event.get('severity', 0) > 70,
            event.get('event_type') in ['privilege_escalation', 'data_exfiltration'],
            self.is_anomalous(event),
            self.matches_threat_pattern(event)
        ]
        
        return any(triggers)
    
    def is_anomalous(self, event: Dict) -> bool:
        """Check for anomalies"""
        return False  # Override in subclasses
    
    def matches_threat_pattern(self, event: Dict) -> bool:
        """Check threat patterns"""
        return False  # Override in subclasses
    
    def quick_decision(self, event: Dict) -> Dict:
        """Fast heuristic decision"""
        return {
            'agent': self.agent_type.value,
            'event_id': event.get('event_id'),
            'verdict': 'benign',
            'confidence': 0.6,
            'reasoning': 'Low severity, no threat indicators',
            'action': 'monitor'
        }
    
    def timeout_decision(self, event: Dict) -> Dict:
        """Decision when analysis times out"""
        return {
            'agent': self.agent_type.value,
            'event_id': event.get('event_id'),
            'verdict': 'suspicious',
            'confidence': 0.3,
            'reasoning': 'Analysis timeout - requires manual review',
            'action': 'investigate'
        }
    
    async def cognitive_analysis(self, event: Dict) -> Dict:
        """LLM-powered cognitive reasoning"""
        
        # Build analysis prompt
        prompt = self.build_cognitive_prompt(event)
        
        # Generate reasoning
        inputs = self.llm['tokenizer'](
            prompt,
            return_tensors='pt',
            truncation=True,
            max_length=512
        ).to('cuda')
        
        with torch.cuda.amp.autocast():
            outputs = self.llm['model'].generate(
                **inputs,
                max_new_tokens=self.config.llm_max_tokens,
                temperature=0.3,
                do_sample=False,
                pad_token_id=self.llm['tokenizer'].eos_token_id
            )
        
        # Parse reasoning
        reasoning = self.llm['tokenizer'].decode(
            outputs[0][inputs['input_ids'].shape[1]:],
            skip_special_tokens=True
        )
        
        # Extract decision
        return self.parse_reasoning(reasoning, event)
    
    def build_cognitive_prompt(self, event: Dict) -> str:
        """Build prompt for cognitive analysis"""
        
        # Get relevant context
        context = self.get_relevant_context(event)
        
        prompt = f"""You are a {self.agent_type.value} analyzing a security event.

Event Details:
- Type: {event.get('event_type')}
- Severity: {event.get('severity')}
- Source: {event.get('source_ip')}
- Destination: {event.get('destination_ip')}
- Process: {event.get('process_name')}

Recent Context:
{context}

Analyze this event and determine:
1. Is this malicious? (yes/no)
2. Confidence level (0-1)
3. Brief reasoning (max 50 words)
4. Recommended action

Response format:
VERDICT: [malicious/benign/suspicious]
CONFIDENCE: [0.0-1.0]
REASONING: [brief explanation]
ACTION: [block/monitor/allow/investigate]"""
        
        return prompt
    
    def get_relevant_context(self, event: Dict) -> str:
        """Get relevant context for analysis"""
        recent_events = list(self.context_window)[-5:]
        context_lines = []
        
        for e in recent_events:
            context_lines.append(
                f"- {e.get('event_type')} from {e.get('source_ip')} "
                f"(severity: {e.get('severity')})"
            )
        
        return '\n'.join(context_lines) if context_lines else "No recent context"
    
    def parse_reasoning(self, reasoning: str, event: Dict) -> Dict:
        """Parse LLM output into structured decision"""
        
        lines = reasoning.strip().split('\n')
        decision = {
            'agent': self.agent_type.value,
            'event_id': event.get('event_id'),
            'timestamp': time.time()
        }
        
        # Parse structured output
        for line in lines:
            if line.startswith('VERDICT:'):
                decision['verdict'] = line.split(':', 1)[1].strip().lower()
            elif line.startswith('CONFIDENCE:'):
                try:
                    decision['confidence'] = float(line.split(':', 1)[1].strip())
                except:
                    decision['confidence'] = 0.5
            elif line.startswith('REASONING:'):
                decision['reasoning'] = line.split(':', 1)[1].strip()
            elif line.startswith('ACTION:'):
                decision['action'] = line.split(':', 1)[1].strip().lower()
        
        # Validate decision
        if 'verdict' not in decision:
            decision['verdict'] = 'suspicious'
        if 'confidence' not in decision:
            decision['confidence'] = 0.5
        
        return decision
    
    def update_metrics(self, latency: float, decision: Dict):
        """Update agent metrics"""
        self.metrics['decisions_made'] += 1
        self.metrics['avg_confidence'] = (
            (self.metrics['avg_confidence'] * (self.metrics['decisions_made'] - 1) +
             decision.get('confidence', 0)) / self.metrics['decisions_made']
        )
        self.metrics['avg_latency_ms'] = (
            (self.metrics['avg_latency_ms'] * (self.metrics['decisions_made'] - 1) +
             latency * 1000) / self.metrics['decisions_made']
        )

    def pattern_matches(self, event: Dict, pattern: Dict) -> bool:
        """Fuzzy pattern matching"""
        
        matches = 0
        total = len(pattern['conditions'])
        
        for condition in pattern['conditions']:
            field = condition['field']
            operator = condition['operator']
            value = condition['value']
            
            event_value = event.get(field)
            
            if operator == 'equals' and event_value == value:
                matches += 1
            elif operator == 'contains' and value in str(event_value):
                matches += 1
            elif operator == 'greater' and event_value > value:
                matches += 1
        
        # Fuzzy matching threshold
        return (matches / total) > 0.7
```

```python
class ConsensusEngine:
    """Multi-agent consensus mechanism"""
    
    def __init__(self):
        self.strategies = {
            'conservative': self.conservative_consensus,
            'majority': self.majority_consensus,
            'weighted': self.weighted_consensus
        }
        self.strategy = 'conservative'  # Default
    
    def evaluate(self, decisions: List[Dict]) -> Dict:
        """Evaluate agent decisions"""
        
        if not decisions:
            return {
                'verdict': 'unknown',
                'confidence': 0.0,
                'reasoning': 'No agent decisions available'
            }
        
        return self.strategies[self.strategy](decisions)
    
    def conservative_consensus(self, decisions: List[Dict]) -> Dict:
        """Any high-confidence threat triggers alert"""
        
        # Check for any malicious verdict
        threats = [d for d in decisions if d['verdict'] == 'malicious']
        
        if threats:
            # Return highest confidence threat
            best_threat = max(threats, key=lambda x: x['confidence'])
            best_threat['consensus_type'] = 'conservative'
            best_threat['agent_votes'] = len(threats)
            return best_threat
        
        # Check for suspicious
        suspicious = [d for d in decisions if d['verdict'] == 'suspicious']
        
        if len(suspicious) >= 2:  # Multiple agents suspicious
            avg_confidence = sum(d['confidence'] for d in suspicious) / len(suspicious)
            return {
                'verdict': 'suspicious',
                'confidence': avg_confidence,
                'consensus_type': 'conservative',
                'agent_votes': len(suspicious),
                'reasoning': 'Multiple agents flagged as suspicious'
            }
        
        # Default to benign
        benign = [d for d in decisions if d['verdict'] == 'benign']
        
        if benign:
            avg_confidence = sum(d['confidence'] for d in benign) / len(benign)
            return {
                'verdict': 'benign',
                'confidence': avg_confidence,
                'consensus_type': 'conservative',
                'agent_votes': len(benign),
                'reasoning': 'Consensus: benign'
            }
        
        return {
            'verdict': 'unknown',
            'confidence': 0.0,
            'consensus_type': 'conservative',
            'reasoning': 'Mixed verdicts, no consensus'
        }
    
    def majority_consensus(self, decisions: List[Dict]) -> Dict:
        """Majority vote consensus"""
        verdict_counts = Counter(d['verdict'] for d in decisions)
        majority_verdict = verdict_counts.most_common(1)[0][0]
        
        matching = [d for d in decisions if d['verdict'] == majority_verdict]
        avg_confidence = sum(d['confidence'] for d in matching) / len(matching)
        
        return {
            'verdict': majority_verdict,
            'confidence': avg_confidence,
            'consensus_type': 'majority',
            'agent_votes': len(matching),
            'total_agents': len(decisions)
        }
    
    def weighted_consensus(self, decisions: List[Dict]) -> Dict:
        """Weighted by agent confidence"""
        verdict_scores = {}
        
        for decision in decisions:
            verdict = decision['verdict']
            confidence = decision['confidence']
            
            if verdict not in verdict_scores:
                verdict_scores[verdict] = 0
            verdict_scores[verdict] += confidence
        
        best_verdict = max(verdict_scores, key=verdict_scores.get)
        
        return {
            'verdict': best_verdict,
            'confidence': verdict_scores[best_verdict] / len(decisions),
            'consensus_type': 'weighted',
            'scores': verdict_scores
        }

    """Manages agent lifecycle and coordination"""

        self.agents = {}

        self.consensus = ConsensusEngine()

        """Initialize agent pool"""
        
        agent_count = self.agent_configs[self.tier]
        
        # Create network analysts
        for i in range(agent_count['network']):
            agent = NetworkAnalystAgent(AgentConfig(tier=self.tier))
            self.agents[f'network_{i}'] = agent
        
        # Create threat hunters
        for i in range(agent_count['threat']):
            agent = ThreatHunterAgent(AgentConfig(tier=self.tier))
            self.agents[f'threat_{i}'] = agent
        
        # Create incident responders
        for i in range(agent_count['incident']):
            agent = IncidentResponderAgent(AgentConfig(tier=self.tier))
            self.agents[f'incident_{i}'] = agent

        await self.message_bus.start()

        # Check resource availability
        if not self.resource_manager.can_process():
            return {'verdict': 'queued', 'reason': 'resource_limit'}
        
        # Select relevant agents
        selected_agents = self.select_agents(event)
        
        # Parallel analysis
        tasks = []
        for agent_id in selected_agents:
            agent = self.agents[agent_id]
            tasks.append(agent.analyze(event))
        
        # Gather decisions with timeout
        try:
            async with asyncio.timeout(10):
                decisions = await asyncio.gather(*tasks, return_exceptions=True)
        except asyncio.TimeoutError:
            decisions = []
        
        # Filter valid decisions
        valid_decisions = [
            d for d in decisions 
            if isinstance(d, dict) and 'verdict' in d
        ]
        
        # Consensus decision
        final_decision = self.consensus.evaluate(valid_decisions)
        
        # Execute response if needed
        if final_decision['verdict'] == 'malicious':
            responder = self.get_responder()
            if responder:

        return final_decision

        selected = []

        if severity > 50 or event_type in ['auth_failure', 'privilege_escalation']:
            selected.append('threat_0')
        
        # Add incident responder for high severity
        if severity > 70:
            selected.append('incident_0')
        
        # Load balance if multiple agents available
        if self.tier != 'validation':
            selected = self.load_balance(selected)
        
        return selected
    
    def load_balance(self, agents: List[str]) -> List[str]:
        """Balance load across available agents"""
        
        balanced = []
        
        for agent_type in ['network', 'threat', 'incident']:
            type_agents = [a for a in agents if a.startswith(agent_type)]
            
            if type_agents:
                # Select least loaded agent of this type
                loads = {
                    a: self.agents[a].metrics['decisions_made'] 
                    for a in self.agents 
                    if a.startswith(agent_type)
                }
                
                least_loaded = min(loads, key=loads.get)
                balanced.append(least_loaded)
        
        return balanced
    
    def get_responder(self) -> Optional[IncidentResponderAgent]:
        """Get available incident responder"""
        responders = [
            agent for name, agent in self.agents.items()
            if name.startswith('incident')
        ]
        
        return responders[0] if responders else None
```

```python
class SIEMConnector(ABC):
    """Abstract base for SIEM integrations"""
    
    @abstractmethod
    async def send_event(self, event: Dict) -> bool:
        pass

class TokenBucket:
    """Rate limiting implementation"""
    
    def __init__(self, rate: int, capacity: int):
        self.rate = rate
        self.capacity = capacity
        self.tokens = capacity
        self.last_update = time.time()
    
    def consume(self, tokens: int = 1) -> bool:
        """Try to consume tokens"""
        now = time.time()
        elapsed = now - self.last_update
        
        # Add new tokens
        self.tokens = min(self.capacity, self.tokens + elapsed * self.rate)
        self.last_update = now
        
        # Try to consume
        if self.tokens >= tokens:
            self.tokens -= tokens
            return True
        
        return False

class SyslogConnector(SIEMConnector):
    """RFC 5424 compliant syslog sender"""
    
    def __init__(self, host: str, port: int = 514):
        self.host = host
        self.port = port
        self.facility = syslog.LOG_LOCAL0
        
        # Rate limiting
        self.rate_limiter = TokenBucket(
            rate=100,  # events per second
            capacity=1000
        )
    
    def format_rfc5424(self, event: Dict) -> str:
        """Format event as RFC 5424 syslog"""
        
        # Calculate priority
        severity = self.map_severity(event['severity'])
        priority = self.facility * 8 + severity
        
        # Timestamp
        timestamp = datetime.now().isoformat()
        
        # Structured data
        structured = f'[neurons@12345 ' \
                    f'eventType="{event["event_type"]}" ' \
                    f'severity="{event["severity"]}" ' \
                    f'verdict="{event.get("verdict", "unknown")}" ' \
                    f'confidence="{event.get("confidence", 0)}"]'
        
        # Message
        msg = json.dumps({
            'event_id': event['event_id'],
            'source': event.get('source_ip'),
            'destination': event.get('destination_ip'),
            'details': event.get('metadata', {})
        })
        
        return f'<{priority}>1 {timestamp} {socket.gethostname()} ' \
               f'NEURONS - - {structured} {msg}'
    
    def map_severity(self, severity: float) -> int:
        """Map NEURONS severity to syslog severity"""
        
        if severity >= 90:
            return syslog.LOG_CRIT
        elif severity >= 70:
            return syslog.LOG_ERR
        elif severity >= 50:
            return syslog.LOG_WARNING
        elif severity >= 30:
            return syslog.LOG_NOTICE
        else:
            return syslog.LOG_INFO
    
    async def send_event(self, event: Dict) -> bool:
        """Send event via syslog"""
        
        if not self.rate_limiter.consume():
            return False
        
        try:
            message = self.format_rfc5424(event)
            
            # Send UDP
            sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
            sock.sendto(message.encode(), (self.host, self.port))
            sock.close()
            
            return True
            
        except Exception as e:
            logger.error(f"Syslog send failed: {e}")
            return False
```

```python
                spikes = torch.stack(spk_rec, dim=0)
                
                # Predict from spike patterns
                prediction = self.predictor(spikes.mean(dim=0))
                
                return F.softmax(prediction, dim=-1)
        
        return PredictiveSNN(n_neurons=self.config['snn_neurons'])
    
    async def predict(self, features: np.ndarray) -> Dict:
        """Predict attack using SNN"""
        
        # Convert to tensor
        x = torch.FloatTensor(features)
        
        # Run prediction
        with torch.no_grad():
            output = self.model(x)
        
        # Parse output
        probs = output.cpu().numpy()[0]
        
        prediction = {
            'method': 'snn',
            'no_threat_prob': float(probs[0]),
            'imminent_prob': float(probs[1]),
            'active_prob': float(probs[2]),
            'threat_detected': probs[1] + probs[2] > self.spike_threshold
        }
        
        # Estimate time to attack
        if prediction['threat_detected']:
            if probs[2] > 0.5:  # Active attack
                prediction['time_to_attack_ms'] = 0
            elif probs[1] > 0.5:  # Imminent
                prediction['time_to_attack_ms'] = 50 + (1 - probs[1]) * 100
            else:
                prediction['time_to_attack_ms'] = 150
        
        return prediction

class PatternPredictor:
    """Pattern-based attack prediction"""
    
    def __init__(self, tier='validation'):
        self.tier = tier
        self.patterns = self.load_attack_patterns()
        self.pattern_matcher = PatternMatcher()
    
    def load_attack_patterns(self) -> List[Dict]:
        """Load known attack patterns"""
        
        return [
            {
                'name': 'ransomware_precursor',

                'threshold': 0.75,

            },

            },

            }
        ]
    
    async def predict(self, features: np.ndarray) -> Dict:
        """Predict based on pattern matching"""
        
        # Match against known patterns
        matches = []
        
        for pattern in self.patterns:
            similarity = self.pattern_matcher.match(features, pattern['signature'])
            
            if similarity > pattern['threshold']:
                matches.append({
                    'pattern': pattern['name'],
                    'similarity': similarity,
                    'time_to_attack': pattern['typical_time_ms']
                })
        
        if matches:
            best_match = max(matches, key=lambda x: x['similarity'])
            
            return {
                'method': 'pattern',
                'threat_detected': True,
                'pattern_matched': best_match['pattern'],
                'confidence': best_match['similarity'],
                'time_to_attack_ms': best_match['time_to_attack']
            }
        
        return {
            'method': 'pattern',
            'threat_detected': False,
            'confidence': 0.0
        }

class PatternMatcher:
    """Pattern matching utilities"""
    
    def match(self, features: np.ndarray, pattern: np.ndarray) -> float:
        """Calculate pattern similarity"""
        
        # Flatten if needed
        features = features.flatten()
        pattern = pattern.flatten()
        
        # Ensure same size
        min_len = min(len(features), len(pattern))
        features = features[:min_len]
        pattern = pattern[:min_len]
        
        # Cosine similarity
        dot_product = np.dot(features, pattern)
        norm_features = np.linalg.norm(features)
        norm_pattern = np.linalg.norm(pattern)
        
        if norm_features == 0 or norm_pattern == 0:
            return 0.0
        
        similarity = dot_product / (norm_features * norm_pattern)
        
        # Normalize to [0, 1]
        return (similarity + 1) / 2

class SequencePredictor:
    """Sequence-based attack prediction"""
    
    def __init__(self, tier='validation'):
        self.tier = tier
        self.model = self.build_lstm_model()
    
    def build_lstm_model(self):
        """Build LSTM for sequence prediction"""
        
        class SequenceLSTM(nn.Module):
            def __init__(self):
                super().__init__()
                
                self.lstm = nn.LSTM(
                    input_size=256,
                    hidden_size=128,
                    num_layers=2,
                    batch_first=True,
                    dropout=0.1
                )
                
                self.classifier = nn.Sequential(
                    nn.Linear(128, 64),
                    nn.ReLU(),
                    nn.Dropout(0.1),
                    nn.Linear(64, 2)  # [benign, threat]
                )
            
            def forward(self, x):
                # LSTM processing
                lstm_out, _ = self.lstm(x.unsqueeze(1))
                
                # Take last output
                last_output = lstm_out[:, -1, :]
                
                # Classify
                output = self.classifier(last_output)
                
                return F.softmax(output, dim=-1)
        
        return SequenceLSTM()
    
    async def predict(self, features: np.ndarray) -> Dict:
        """Predict using sequence analysis"""
        
        x = torch.FloatTensor(features)
        
        with torch.no_grad():
            output = self.model(x)
        
        probs = output.cpu().numpy()[0]
        
        return {
            'method': 'sequence',
            'threat_detected': probs[1] > 0.6,
            'confidence': float(probs[1]),
            'time_to_attack_ms': int(100 * (1 - probs[1]) + 50) if probs[1] > 0.6 else 0
        }

class PredictionFusion:
    """Fuse multiple predictions"""
    
    def fuse(self, predictions: List[Dict]) -> Dict:
        """Combine predictions from multiple models"""
        
        # Extract threat probabilities
        threat_scores = []
        time_estimates = []
        
        for pred in predictions:
            if pred.get('threat_detected'):
                # Weight by method
                weight = {
                    'snn': 0.5,
                    'pattern': 0.3,
                    'sequence': 0.2
                }.get(pred.get('method'), 0.1)
                
                threat_scores.append(
                    pred.get('confidence', pred.get('imminent_prob', 0)) * weight
                )
                
                if 'time_to_attack_ms' in pred:
                    time_estimates.append(pred['time_to_attack_ms'])
        
        if not threat_scores:
            return {
                'threat_detected': False,
                'fusion_confidence': 0.0,
                'prediction_window_ms': 0
            }
        
        # Weighted average
        fusion_score = sum(threat_scores)
        
        # Time estimate (conservative - use minimum)
        time_to_attack = min(time_estimates) if time_estimates else 100
        
        return {
            'threat_detected': fusion_score > 0.5,
            'fusion_confidence': min(fusion_score, 1.0),
            'prediction_window_ms': int(time_to_attack),
            'contributing_methods': [p['method'] for p in predictions if p.get('threat_detected')]
        }
```

```python
        self.engine = PredictiveEngine(tier)

        self.windows = {
            'micro': deque(maxlen=100),    # Last 100 events
            'short': deque(maxlen=1000),   # Last 1000 events
            'medium': deque(maxlen=10000)  # Last 10000 events
        }

        # Alert management
        self.alert_manager = AlertManager()
    
    async def monitor_stream(self, event_stream):
        """Monitor event stream with predictions"""
        
        async for event in event_stream:
            # Add to windows
            for window in self.windows.values():
                window.append(event)
            
            # Check if prediction needed
            if self.should_predict(event):
                # Run prediction on appropriate window
                window = self.select_window(event)
                prediction = await self.engine.predict(list(window))
                
                # Cache prediction
                self.prediction_cache[event['event_id']] = prediction
                
                # Generate alert if needed
                if prediction['threat_detected']:
                    await self.generate_predictive_alert(prediction, event)
    
    def should_predict(self, event: Dict) -> bool:
        """Determine if prediction should run"""
        
        triggers = [
            event.get('severity', 0) > 50,
            event.get('event_type') in ['auth_failure', 'scan_detected'],
            self.is_velocity_increasing(),
            self.is_pattern_emerging()
        ]
        
        return any(triggers)
    
    def select_window(self, event: Dict) -> deque:
        """Select appropriate window for prediction"""
        
        severity = event.get('severity', 0)
        
        if severity > 80:
            return self.windows['micro']  # Use recent events for high severity
        elif severity > 50:
            return self.windows['short']
        else:
            return self.windows['medium']
    
    def is_velocity_increasing(self) -> bool:
        """Check if event velocity is increasing"""
        
        if len(self.windows['micro']) < 50:
            return False
        
        # Calculate velocity over last 10 vs previous 10
        recent = list(self.windows['micro'])[-10:]
        previous = list(self.windows['micro'])[-20:-10]
        
        if not recent or not previous:
            return False
        
        recent_rate = len(recent) / (recent[-1]['timestamp'] - recent[0]['timestamp'] + 0.001)
        previous_rate = len(previous) / (previous[-1]['timestamp'] - previous[0]['timestamp'] + 0.001)
        
        return recent_rate > previous_rate * 1.5
    
    def is_pattern_emerging(self) -> bool:
        """Check for emerging attack patterns"""
        
        if len(self.windows['short']) < 100:
            return False
        
        # Count event types in recent window
        recent_events = list(self.windows['short'])[-100:]
        event_types = [e.get('event_type') for e in recent_events]
        
        # Check for suspicious patterns
        suspicious_types = ['auth_failure', 'privilege_escalation', 'scan_detected']
        suspicious_count = sum(1 for t in event_types if t in suspicious_types)
        
        return suspicious_count > 10  # More than 10% suspicious

            'confidence': prediction['fusion_confidence'],

            'recommended_actions': self.get_recommended_actions(prediction),
            'timestamp': time.time()
        }
        
        await self.alert_manager.send_alert(alert)
```

## Knowledge Graph Integration

The knowledge graph integration transforms isolated security events into a connected intelligence network. Think of it as creating a living map of your attack surface that evolves with each event. Here's how NEURONS builds and leverages this knowledge graph:

```python
class SecurityKnowledgeGraph:
    """
    Dynamic knowledge graph for security intelligence
    """
    
    def __init__(self):
        self.graph = nx.DiGraph()
        self.embeddings = {}  # Vector representations for similarity
        
    def build_entity_relationships(self, event: Dict):
        """
        Construct relationships between security entities
        """
        # Create nodes for entities
        entities = {
            'ip': event.get('source_ip'),
            'process': event.get('process_name'),
            'user': event.get('username'),
            'file': event.get('file_path'),
            'technique': self.map_to_mitre(event)
        }
        
        # Add nodes with properties
        for entity_type, entity_value in entities.items():
            if entity_value:
                self.graph.add_node(
                    entity_value,
                    type=entity_type,
                    first_seen=time.time(),
                    risk_score=0
                )
        
        # Create edges based on relationships
        if entities['ip'] and entities['process']:
            self.graph.add_edge(
                entities['ip'],
                entities['process'],
                relationship='executed',
                timestamp=event['timestamp']
            )
        
        # Update risk propagation
        self.propagate_risk(entities)
    
    def propagate_risk(self, entities: Dict):
        """
        Propagate risk scores through the graph
        """
        # Use PageRank-like algorithm for risk propagation
        for entity in entities.values():
            if entity and entity in self.graph:
                neighbors = list(self.graph.neighbors(entity))
                
                # Calculate risk based on neighbor relationships
                risk = sum(
                    self.graph.nodes[n].get('risk_score', 0) * 0.3
                    for n in neighbors
                )
                
                self.graph.nodes[entity]['risk_score'] = min(risk, 100)
    
    def find_attack_paths(self, source: str, target: str) -> List[List[str]]:
        """
        Identify potential attack paths in the knowledge graph
        """
        try:
            # Find all simple paths
            paths = list(nx.all_simple_paths(
                self.graph, source, target, cutoff=5
            ))
            
            # Score paths by risk
            scored_paths = []
            for path in paths:
                path_risk = sum(
                    self.graph.nodes[node].get('risk_score', 0)
                    for node in path
                ) / len(path)
                
                scored_paths.append((path, path_risk))
            
            # Return paths sorted by risk
            return [p for p, _ in sorted(scored_paths, key=lambda x: x[1], reverse=True)]
            
        except nx.NetworkXNoPath:
            return []
```
